cmake_minimum_required(VERSION 3.21)
project(DirectMusicProducerRecovery LANGUAGES CXX RC)

add_executable(producer_startup_probe tests/native/producer_startup_probe.cpp)
target_compile_features(producer_startup_probe PRIVATE cxx_std_17)
target_compile_definitions(producer_startup_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(producer_startup_probe PRIVATE user32)
if(MSVC)
  target_compile_options(producer_startup_probe PRIVATE /W4 /utf-8)
endif()

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The reference probes require Windows x86. Configure with -A Win32.")
endif()

add_executable(timeline_window_probe tests/native/timeline_window_probe.cpp)
target_compile_features(timeline_window_probe PRIVATE cxx_std_17)
target_include_directories(timeline_window_probe PRIVATE src)
target_compile_definitions(timeline_window_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(timeline_window_probe PRIVATE ole32 oleaut32 uuid user32)
if(MSVC)
  target_compile_options(timeline_window_probe PRIVATE /W4 /utf-8)
  # Legacy ATL thunks need Windows' DEP thunk emulation. The probe enables
  # DEP permanently at entry, verifies it, and fails before loading originals
  # if that policy cannot be established. Non-windowed targets retain /NXCOMPAT.
  target_link_options(timeline_window_probe PRIVATE /NXCOMPAT:NO)
endif()

add_library(tempo_core STATIC src/tempo/tempo_track.cpp)
target_compile_features(tempo_core PUBLIC cxx_std_17)
target_include_directories(tempo_core PUBLIC src)
add_library(tempo_strip_manager SHARED src/tempo/tempo_dll.cpp src/tempo/tempo_dll.def src/tempo/tempo_pages.rc)
target_link_libraries(tempo_strip_manager PRIVATE tempo_core ole32 oleaut32 uuid gdi32 user32 comctl32)
target_compile_definitions(tempo_strip_manager PRIVATE UNICODE _UNICODE)
set_target_properties(tempo_strip_manager PROPERTIES OUTPUT_NAME TempoStripMgr)
add_executable(tempo_core_compare tests/native/tempo_core_compare.cpp)
target_link_libraries(tempo_core_compare PRIVATE tempo_core)
if(MSVC)
  target_compile_options(tempo_core PRIVATE /W4 /utf-8)
  target_compile_options(tempo_strip_manager PRIVATE /W4 /utf-8)
  target_compile_options(tempo_core_compare PRIVATE /W4 /utf-8)
endif()

add_executable(com_probe tests/native/com_probe.cpp)
add_executable(com_window_probe tests/native/com_probe.cpp)
target_compile_features(com_window_probe PRIVATE cxx_std_17)
target_include_directories(com_window_probe PRIVATE src)
target_compile_definitions(com_window_probe PRIVATE UNICODE _UNICODE PRODUCER_WINDOWED_PROBE)
target_link_libraries(com_window_probe PRIVATE ole32 oleaut32 uuid gdi32 user32 comctl32)
if(MSVC)
  target_compile_options(com_window_probe PRIVATE /W4 /utf-8)
  target_link_options(com_window_probe PRIVATE /NXCOMPAT:NO)
endif()
target_compile_features(com_probe PRIVATE cxx_std_17)
target_include_directories(com_probe PRIVATE src)
target_compile_definitions(com_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(com_probe PRIVATE ole32 oleaut32 uuid gdi32 user32)
if(MSVC)
  target_compile_options(com_probe PRIVATE /W4 /utf-8)
endif()
