target_compile_definitions(timeline_window_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(timeline_window_probe PRIVATE ole32 oleaut32 uuid user32)
if(MSVC)
  target_compile_options(timeline_window_probe PRIVATE /W4 /utf-8)
  # Legacy ATL thunks need Windows' DEP thunk emulation. The probe enables
  # DEP permanently at entry, verifies it, and fails before loading originals
  # if that policy cannot be established. Non-windowed targets retain /NXCOMPAT.
  target_link_options(timeline_window_probe PRIVATE /NXCOMPAT:NO)
