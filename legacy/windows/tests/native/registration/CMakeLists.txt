cmake_minimum_required(VERSION 3.21)
project(ProducerRegistrationCapture LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Registration capture requires Windows x86; use -A Win32")
endif()
add_executable(producer_registration_probe ../producer_registration_probe.cpp)
target_compile_features(producer_registration_probe PRIVATE cxx_std_17)
target_compile_definitions(producer_registration_probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0600)
target_link_libraries(producer_registration_probe PRIVATE advapi32 bcrypt ole32 oleaut32)
if(MSVC)
  target_compile_options(producer_registration_probe PRIVATE /W4 /utf-8)
endif()
