cmake_minimum_required(VERSION 3.21)
project(TimeSignatureObservation LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "TimeSig original ABI requires Windows x86")
endif()
add_executable(time_signature_probe ../time_signature_probe.cpp)
add_executable(time_signature_connection_probe ../time_signature_connection_probe.cpp)
add_executable(time_signature_style_probe ../time_signature_style_probe.cpp)
target_compile_features(time_signature_style_probe PRIVATE cxx_std_17)
target_include_directories(time_signature_style_probe PRIVATE ../../../src)
target_compile_definitions(time_signature_style_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(time_signature_style_probe PRIVATE ole32 oleaut32 uuid)
set_target_properties(time_signature_style_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(time_signature_connection_probe PRIVATE cxx_std_17)
target_include_directories(time_signature_connection_probe PRIVATE ../../../src)
target_compile_definitions(time_signature_connection_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(time_signature_connection_probe PRIVATE ole32 oleaut32 uuid)
set_target_properties(time_signature_connection_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_compile_features(time_signature_probe PRIVATE cxx_std_17)
target_include_directories(time_signature_probe PRIVATE ../../../src)
target_compile_definitions(time_signature_probe PRIVATE UNICODE _UNICODE)
target_link_libraries(time_signature_probe PRIVATE ole32 oleaut32 uuid)
set_target_properties(time_signature_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_subdirectory(../../../src/time_signature time_signature)
# Keep the standalone DLL beside the probe, as Run-TimeSignatureProbe expects.
set_target_properties(time_signature_strip_manager PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}" ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(MSVC)
  target_compile_options(time_signature_probe PRIVATE /W4 /utf-8)
  target_compile_options(time_signature_connection_probe PRIVATE /W4 /utf-8)
  target_compile_options(time_signature_style_probe PRIVATE /W4 /utf-8)
endif()
