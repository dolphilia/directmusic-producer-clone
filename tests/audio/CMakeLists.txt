cmake_minimum_required(VERSION 3.21)
project(ProducerAudioCapture LANGUAGES CXX)
add_executable(producer_loopback loopback.cpp)
target_compile_features(producer_loopback PRIVATE cxx_std_17)
target_compile_definitions(producer_loopback PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(producer_loopback PRIVATE ole32 uuid)
if(MSVC)
  target_compile_options(producer_loopback PRIVATE /W4 /utf-8)
  set_property(TARGET producer_loopback PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
endif()
