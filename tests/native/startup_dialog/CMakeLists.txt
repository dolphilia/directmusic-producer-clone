cmake_minimum_required(VERSION 3.21)
project(ProducerStartupDialog LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This diagnostic requires Windows")
endif()
add_executable(producer_startup_dialog ../producer_startup_dialog.cpp)
target_compile_features(producer_startup_dialog PRIVATE cxx_std_17)
target_compile_definitions(producer_startup_dialog PRIVATE UNICODE _UNICODE)
target_link_libraries(producer_startup_dialog PRIVATE user32 gdi32 gdiplus)
if(MSVC)
  target_compile_options(producer_startup_dialog PRIVATE /W4 /utf-8)
endif()
